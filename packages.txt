# Packages from the official repositories used by these dotfiles. Installed by
# «./install.sh» (pacman -S --needed: the ones already installed are left alone).
# One per line; whatever comes after # is a comment.

# Desktop
hyprland
uwsm                      # session (greetd starts it with uwsm)
quickshell                # the shell (bar, Settings, lock…)
hyprlock                  # fallback lock
hypridle                  # idle lock and suspend
hyprpicker                # color picker (menu)
xdg-desktop-portal-hyprland  # screen sharing (xdph.conf + SharePicker)
greetd                    # autologin (and agreety after logging out)
gnome-keyring             # keyring (greetd PAM, lock and hyprlock)

# Terminal and tools
alacritty
neovim
eza                       # colored ls -l (prompt.sh)
less                      # pager for git and man
git
curl                      # WebApp and TUI icons, fallback cursor (install.sh)
fzf                       # package lists (menu)
jq
btop                      # click on the system monitor (bar)
fastfetch                 # menu → About
pciutils                  # lspci (Settings → System)
python                    # scripts/mouse.py (shell tests)

# Appearance
ttf-jetbrains-mono-nerd
adw-gtk-theme             # base of the NothingOS GTK theme

# Hardware and services
pipewire
pipewire-pulse
wireplumber               # wpctl (volume keys) and the shell
playerctl                 # media keys
upower                    # battery (bar)
power-profiles-daemon     # power profiles (Settings → Power)
brightnessctl             # brightness (scripts/brightness.sh)
ddcutil                   # external monitor brightness (DDC/CI; i2c group)
asusctl                   # ROG Zephyrus G14
rog-control-center        # (autostart.lua)
alsa-utils                # mixer state (G14 audio patch)
networkmanager            # Wi-Fi (nmcli)
bluez
bluez-utils

# Printing (Settings → Printers; it can also install it from there)
cups
cups-filters
ghostscript               # used by the CUPS filters (without it: «gstoraster filter failed»)
avahi                     # find printers on the network
nss-mdns

# Apps
nautilus                  # file manager (programs.lua)
dosfstools                # mkfs.fat: format USB drives as FAT32 (also from Disks)
exfatprogs                # mkfs.exfat: USB drives as exFAT (files over 4 GB)
gnome-disk-utility        # «Disks»: Nautilus opens it from a drive's properties (D-Bus org.gnome.DiskUtility)
zenity
xdg-utils                 # xdg-open
xdg-terminal-exec         # open terminal apps (nvim…) from Nautilus
chromium                  # WebApps
flatpak
spotify-launcher          # (spotify-launcher.conf)

# System menu: screenshots, recording, OCR, clipboard, notifications
grim
slurp
satty
wl-clipboard
libnotify
tesseract
tesseract-data-eng        # OCR; more languages: tesseract-data-<lang>
gpu-screen-recorder
ffmpeg

# Updates, snapshots and boot (arch-update, docs/SYSTEM.md)
pacman-contrib            # paccache and checkupdates
fwupd                     # firmware (menu → Update)
snapper
btrfs-progs
limine
plymouth                  # disabled (plymouth.enable=0): nothing theme kept to try later
sbctl                     # Secure Boot
base-devel                # to build AUR packages (yay)
